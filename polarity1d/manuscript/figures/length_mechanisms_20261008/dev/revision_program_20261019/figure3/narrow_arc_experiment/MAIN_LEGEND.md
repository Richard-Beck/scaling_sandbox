M_n(t)=C(t)\left\langle u(q,t)e^{2\pi i n q}\right\rangle,
\qquad P(t)=\frac{M_1(t)}{M_1(0)},\qquad R(t)=|P(t)|,
\tag{1}

$$
\omega_{\rm deg}(t)=\frac{180}{\pi}
\frac{\operatorname{Im}\!\left[\overline{M_1(t)}\,\dot M_1(t)\right]}{|M_1(t)|^2}.

$$
